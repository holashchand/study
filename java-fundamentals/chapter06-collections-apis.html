6.1 String Class — Immutability & String Pool

Immutability: Every modification creates a new String object

String original = "Hello";
String modified = original.toUpperCase();  // creates new String "HELLO"
System.out.println(original);              // still "Hello"
System.out.println(modified);              // "HELLO"

// Reassignment vs mutation:
String s = "Java";
s = s + " 8";                              // creates new String, reassigns reference
System.out.println(s);                     // "Java 8"
// No method on String modifies the object itself

String Pool: String literals are interned

String a = "hello";        // stored in string pool
String b = "hello";        // retrieves same object from pool
String c = new String("hello");  // creates new object on heap

System.out.println(a == b);       // true (same object in pool)
System.out.println(a == c);       // false (c is on heap)
System.out.println(a.equals(c));  // true (same content)

// WRONG: comparing with ==
if (userInput == "admin") { }  // fails when userInput is new String

// CORRECT: comparing with .equals()
if (userInput.equals("admin")) { }  // works always

The == vs .equals() trap

// WRONG: uses == (compares references)
String input = new String("hello");
if (input == "hello") {
    System.out.println("Match");  // NOT printed (different objects)
}

// CORRECT: uses .equals() (compares values)
String input = new String("hello");
if (input.equals("hello")) {
    System.out.println("Match");  // printed (values are equal)
}

// Intern for forced pool lookup:
String x = new String("hello").intern();
String y = "hello";
System.out.println(x == y);  // true (both now from pool)

6.3 StringBuilder — Mutable String Buffer

Performance comparison: concatenation vs StringBuilder

// WRONG: String concatenation in loop creates O(n²) objects
long start = System.currentTimeMillis();
String result = "";
for (int i = 0; i < 10000; i++) {
    result += i;  // creates new String each time: "", "0", "01", "012", ...
}
long end = System.currentTimeMillis();
System.out.println("Concat time: " + (end - start) + "ms");  // ~50-100ms

// CORRECT: StringBuilder is O(n)
start = System.currentTimeMillis();
StringBuilder sb = new StringBuilder();
for (int i = 0; i < 10000; i++) {
    sb.append(i);  // amortized O(1) per append
}
String result = sb.toString();
end = System.currentTimeMillis();
System.out.println("StringBuilder time: " + (end - start) + "ms");  // <1ms

StringBuilder methods: append, insert, delete

StringBuilder sb = new StringBuilder("Hello");

// append(): add to end
sb.append(" ");
sb.append("World");
System.out.println(sb);  // "Hello World"

// Can chain:
sb = new StringBuilder()
    .append("Java ")
    .append(8)
    .append(" is ")
    .append("awesome");
System.out.println(sb);  // "Java 8 is awesome"

// insert(): add at index
sb = new StringBuilder("Hello World");
sb.insert(5, " Java");
System.out.println(sb);  // "Hello Java World"

// delete(): remove range [start, end)
sb = new StringBuilder("Hello World");
sb.delete(5, 11);
System.out.println(sb);  // "Hello"

// deleteCharAt(): remove single char
sb = new StringBuilder("Hello");
sb.deleteCharAt(1);
System.out.println(sb);  // "Hllo"

replace, reverse, charAt, setCharAt

StringBuilder sb = new StringBuilder("Hello World");

// replace(): replace substring in range
sb.replace(0, 5, "Hi");
System.out.println(sb);  // "Hi World"

// reverse():
sb = new StringBuilder("Java");
sb.reverse();
System.out.println(sb);  // "avaJ"

// charAt(): get char at index
sb = new StringBuilder("Hello");
System.out.println(sb.charAt(1));  // 'e'

// setCharAt(): set char at index
sb.setCharAt(0, 'J');
System.out.println(sb);  // "Jello"

length(), capacity(), toString()

StringBuilder sb = new StringBuilder("Hello");

System.out.println(sb.length());      // 5 (number of characters)
System.out.println(sb.capacity());    // 21 (internal buffer size)

// Capacity grows as needed:
StringBuilder sb2 = new StringBuilder(5);
sb2.append("Hello");
System.out.println(sb2.capacity());   // 5
sb2.append("World");
System.out.println(sb2.capacity());   // 22 (doubled + 2)

// toString(): convert to immutable String
String result = sb.toString();
System.out.println(result instanceof String);  // true

StringBuilder vs StringBuffer (thread safety)

// StringBuilder — not synchronized, faster
StringBuilder sb = new StringBuilder();
sb.append("Hello").append("World");  // ~1μs

// StringBuffer — synchronized, slower (for thread-safe code)
StringBuffer sbf = new StringBuffer();
sbf.append("Hello").append("World");  // ~10μs (locking overhead)

// Single-threaded code: use StringBuilder always
// Multi-threaded (if needed): use StringBuffer or external synchronization

6.6 Collections Framework — Hierarchy

Collection hierarchy diagram

                    Iterable
                       │
                  Collection
                 /    |    \
               List  Set   Queue
               │      │      │
          ArrayList HashSet ArrayDeque
          LinkedList LinkedHashSet PriorityQueue
          Vector   TreeSet  LinkedList
          Stack

         Map (separate hierarchy)
        /    |     \
     HashMap LinkedHashMap TreeMap
     Hashtable ConcurrentHashMap WeakHashMap

Quick reference: When to use each

// List — ordered, allows duplicates, index access
List<String> list = new ArrayList<>();      // most common, O(1) access
List<String> list = new LinkedList<>();     // frequent add/remove at ends

// Set — unique values only, no order
Set<String> set = new HashSet<>();          // O(1) operations, no order
Set<String> set = new LinkedHashSet<>();    // insertion order preserved
Set<String> set = new TreeSet<>();          // sorted order, O(log n)

// Map — key-value pairs
Map<String, Integer> map = new HashMap<>();         // O(1) average
Map<String, Integer> map = new TreeMap<>();        // sorted by key
Map<String, Integer> map = new LinkedHashMap<>();  // insertion order

// Queue — FIFO
Queue<String> queue = new LinkedList<>();  // general queue
Queue<Integer> pq = new PriorityQueue<>(); // heap-based ordering

// Deque — double-ended
Deque<String> deque = new ArrayDeque<>();  // more efficient than LinkedList

Key principle: Program to interface, not implementation

// CORRECT: use interface type
List<String> list = new ArrayList<>();
Set<Integer> set = new HashSet<>();
Map<String, Integer> map = new HashMap<>();

// Method signature should accept interface:
public void processItems(List<String> items) { }  // flexible
public void processItems(ArrayList<String> items) { }  // too restrictive

// Call works with any implementation:
processItems(new ArrayList<>());   // works
processItems(new LinkedList<>());  // works (if parameter is List)

6.14 Comparable vs Comparator

Comparable — natural ordering implemented by class

class Student implements Comparable<Student> {
    String name;
    int gpa;

    Student(String name, int gpa) {
        this.name = name;
        this.gpa = gpa;
    }

    // Natural ordering: by GPA (ascending)
    @Override
    public int compareTo(Student other) {
        // return negative if this < other
        // return 0 if this == other
        // return positive if this > other
        return Integer.compare(this.gpa, other.gpa);
    }

    @Override
    public String toString() {
        return name + " (" + gpa + ")";
    }
}

List<Student> students = new ArrayList<>(Arrays.asList(
    new Student("Alice", 3),
    new Student("Bob", 4),
    new Student("Charlie", 2)
));

Collections.sort(students);  // uses compareTo()
System.out.println(students);
// Charlie (2), Alice (3), Bob (4)

class Student {
    String name;
    int gpa;

    Student(String name, int gpa) {

        this.gpa = gpa;

    @Override
    public String toString() {

    new Student("Alice", 4)

6.15 Iterator & ListIterator

Iterator — safe removal during iteration

List<String> list = new ArrayList<>(List.of("a", "b", "c", "d", "e"));

// WRONG: ConcurrentModificationException
for (String s : list) {
    if (s.equals("b")) {
        list.remove(s);  // throws ConcurrentModificationException
    }
}

// CORRECT: use iterator.remove()
Iterator<String> it = list.iterator();
while (it.hasNext()) {
    String s = it.next();
    if (s.equals("b")) {
        it.remove();  // safe removal
    }
}
System.out.println(list);  // [a, c, d, e]

// CORRECT: use removeIf()
list = new ArrayList<>(List.of("a", "b", "c", "d", "e"));
list.removeIf(s -> s.equals("b"));
System.out.println(list);  // [a, c, d, e]

ListIterator — bidirectional iteration

List<String> list = new ArrayList<>(List.of("a", "b", "c", "d"));

// Forward iteration:
ListIterator<String> lit = list.listIterator();
while (lit.hasNext()) {
    System.out.println(lit.next());
}
// Output: a, b, c, d

// Backward iteration (start from end):
lit = list.listIterator(list.size());
while (lit.hasPrevious()) {
    System.out.println(lit.previous());
}
// Output: d, c, b, a

// Get current index:
lit = list.listIterator();
lit.next();  // a
lit.next();  // b
System.out.println(lit.previousIndex());  // 1 (index of b)
System.out.println(lit.nextIndex());      // 2 (index of c)

// Add elements during iteration:
lit = list.listIterator();
lit.next();  // a
lit.add("X");  // insert X after a
System.out.println(list);  // [a, X, b, c, d]

// Set element (replace):
lit = list.listIterator();
lit.next();  // a
lit.set("A");  // replace a with A
System.out.println(list);  // [A, X, b, c, d]

Iterator vs for-each performance

List<String> list = new ArrayList<>(List.of("a", "b", "c", "d", "e"));

// For-each (uses iterator internally):
for (String s : list) {
    System.out.println(s);
}

// Explicit iterator (same performance):
for (Iterator<String> it = list.iterator(); it.hasNext(); ) {
    String s = it.next();
    System.out.println(s);
}

// Index-based (slower for LinkedList):
for (int i = 0; i < list.size(); i++) {
    String s = list.get(i);
    System.out.println(s);
}

// For ArrayList: index-based is fine (O(1) access)
// For LinkedList: use iterator (O(n) vs O(n²))

List<String> list = new ArrayList<>(List.of("a", "b", "c", "d"));

// WRONG: structural modification during iteration

    for (String s : list) {

    String s = it.next();
